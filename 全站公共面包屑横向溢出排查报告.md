   1 | /* =========================================================

  12 |    4. 支持两种常用视觉：
  13 |       - bar：Banner 下方整条白底面包屑，适合资源中心、关于我们

  15 | 
  16 |    注意：
  17 |    1. 这个组件只负责展示结构

  20 | ========================================================= */
  21 | 
  22 | import Link from "next/link";
  23 | 
  24 | import styles from "./SiteBreadcrumb.module.css";
  25 | 
  26 | export type SiteBreadcrumbItem = {
  27 |   label: string;
  28 |   href?: string;
  29 | };
  30 | 
  31 | type SiteBreadcrumbVariant = "bar" | "inline";
  32 | 
  33 | type SiteBreadcrumbProps = {
  34 |   items: SiteBreadcrumbItem[];
  35 |   ariaLabel?: string;
  36 |   variant?: SiteBreadcrumbVariant;
  37 |   className?: string;
  38 | };
  39 | 
  40 | export default function SiteBreadcrumb({
  41 |   items,
  42 |   ariaLabel = "Breadcrumb",
  43 |   variant = "bar",
  44 |   className = "",
  45 | }: SiteBreadcrumbProps) {
  46 |   if (!items.length) {
  47 |     return null;
  48 |   }
  49 | 
  50 |   return (
  51 |     <nav
  52 |       className={`${styles.breadcrumb} ${styles[variant]} ${className}`}
  53 |       aria-label={ariaLabel}
  54 |     >
  55 |       <div className={styles.inner}>
  56 |         {items.map((item, index) => {
  57 |           const isLast = index === items.length - 1;
  58 | 
  59 |           return (
  60 |             <span className={styles.item} key={`${item.label}-${index}`}>
  61 |               {item.href && !isLast ? (
  62 |                 <Link href={item.href}>{item.label}</Link>
  63 |               ) : (
  64 |                 <strong>{item.label}</strong>
  65 |               )}
  66 | 
  67 |               {!isLast ? (
  68 |                 <span className={styles.separator} aria-hidden="true">
  69 |                   /
  70 |                 </span>
  71 |               ) : null}
  72 |             </span>
